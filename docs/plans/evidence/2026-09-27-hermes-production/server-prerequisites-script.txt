"""Bounded prerequisites for aicowork@10.0.30.10; preview unless --apply."""

import argparse
import datetime
import fcntl
import hashlib
import json
import os
import stat
import subprocess
import sys
import tempfile
import time
from pathlib import Path

HOME_UUID = "eb129d35-1f52-4577-99e5-25daa322febc"
LUKS_UUID = "8f5a57cb-f1d7-4eb7-8d1f-266d65f8cff6"
MASK = Path("/etc/systemd/zram-generator.conf")
REPORT_DIR = Path("/var/lib/hermes-production-prerequisites")
REPORT = REPORT_DIR / "report.json"


def require(value, reason):
    if not value:
        raise RuntimeError(reason)


def run(args, *, check=True, timeout=60):
    result = subprocess.run(
        args,
        capture_output=True,
        text=True,
        timeout=timeout,
        env={"PATH": "/usr/sbin:/usr/bin:/sbin:/bin", "LANG": "C.UTF-8"},
        stdin=subprocess.DEVNULL,
    )
    if check:
        require(result.returncode == 0, "command-failed:" + Path(args[0]).name)
    return result


def root_directory(path):
    info = path.lstat()
    require(
        stat.S_ISDIR(info.st_mode) and info.st_uid == 0 and not info.st_mode & 0o022,
        "unsafe-root-directory",
    )


def swap_inventory(raw):
    rows = raw.splitlines()
    require(
        rows and rows[0].split() == ["Filename", "Type", "Size", "Used", "Priority"],
        "unexpected-swap-header",
    )
    swaps = []
    for row in rows[1:]:
        fields = row.split()
        require(
            len(fields) == 5 and fields[2].isdigit() and fields[3].isdigit(),
            "unexpected-swap-inventory",
        )
        swaps.append({"name": fields[0], "used": int(fields[3]) * 1024})
    return swaps


def inspect_inputs():
    require(
        run(["findmnt", "-n", "-o", "UUID", "--target", "/home/hermes"]).stdout.strip()
        == HOME_UUID,
        "wrong-target-filesystem",
    )
    for parent in (
        Path("/etc"),
        Path("/etc/systemd"),
        Path("/run"),
        Path("/var"),
        Path("/var/lib"),
    ):
        root_directory(parent)
    for item in Path("/proc/cmdline").read_text().split():
        require(
            item
            not in (
                "systemd.zram",
                "systemd.zram=1",
                "systemd.zram=yes",
                "systemd.zram=true",
                "systemd.zram=on",
            ),
            "kernel-forces-zram",
        )
    for prefix in ("/run", "/etc", "/usr/local/lib", "/usr/lib"):
        directory = Path(prefix) / "systemd/zram-generator.conf.d"
        require(not directory.is_symlink(), "zram-dropin-directory-symlink")
        require(
            not directory.exists() or not any(directory.glob("*.conf")),
            "existing-zram-dropins-require-review",
        )
    require(
        not Path("/run/systemd/zram-generator.conf").exists()
        and not Path("/run/systemd/zram-generator.conf").is_symlink(),
        "runtime-zram-configuration-conflict",
    )
    if MASK.is_symlink():
        require(
            os.readlink(MASK) == "/dev/null" and MASK.lstat().st_uid == 0,
            "existing-zram-mask-conflict",
        )
    elif MASK.exists():
        info = MASK.lstat()
        require(
            stat.S_ISREG(info.st_mode)
            and info.st_uid == 0
            and info.st_nlink == 1
            and not info.st_mode & 0o022
            and MASK.read_bytes() == b"",
            "existing-zram-configuration-conflict",
        )
    swaps = swap_inventory(Path("/proc/swaps").read_text())
    require(all(row["name"] == "/dev/zram0" for row in swaps), "unrelated-swap-requires-review")
    available = next(
        int(line.split()[1]) * 1024
        for line in Path("/proc/meminfo").read_text().splitlines()
        if line.startswith("MemAvailable:")
    )
    require(
        available >= sum(int(row["used"]) for row in swaps) + 2 * 1024**3,
        "insufficient-memory-for-swapoff",
    )
    return swaps


def root_facts():
    facts = {}
    for name, args in {
        "logical_volumes": [
            "lvs",
            "hermes_system",
            "--reportformat",
            "json",
            "--units",
            "b",
            "--nosuffix",
            "-o",
            "vg_name,lv_name,segtype,lv_size",
        ],
        "volume_group": [
            "vgs",
            "hermes_system",
            "--reportformat",
            "json",
            "--units",
            "b",
            "--nosuffix",
            "-o",
            "vg_name,vg_size,vg_free",
        ],
    }.items():
        value = run(args, check=False)
        facts[name] = (
            json.loads(value.stdout) if value.returncode == 0 else {"status": "UNVERIFIED"}
        )
    metadata = run(
        ["cryptsetup", "luksDump", "--dump-json-metadata", "/dev/disk/by-uuid/" + LUKS_UUID],
        check=False,
    )
    if metadata.returncode == 0:
        value = json.loads(metadata.stdout)
        facts["luks2"] = {
            "keyslot_count": len(value.get("keyslots", {})),
            "tpm_tokens": [
                {
                    "pcrs": token.get("tpm2-pcrs"),
                    "bank": token.get("tpm2-pcr-bank"),
                    "pin": token.get("tpm2-pin", False),
                }
                for token in value.get("tokens", {}).values()
                if token.get("type") == "systemd-tpm2"
            ],
        }
    else:
        facts["luks2"] = {"status": "UNVERIFIED"}
    crypttab = Path("/etc/crypttab")
    facts["crypttab"] = []
    if crypttab.exists():
        for line in crypttab.read_text().splitlines():
            fields = line.split()
            if len(fields) >= 2 and fields[1] == "UUID=" + LUKS_UUID:
                facts["crypttab"].append(
                    {
                        "keyfile_bypass": len(fields) > 2 and fields[2] not in ("none", "-"),
                        "tpm2_auto": len(fields) > 3 and "tpm2-device=auto" in fields[3].split(","),
                    }
                )
    modules = run(["lsinitrd", "-m", "/boot/initramfs-" + os.uname().release + ".img"], check=False)
    facts["initramfs_tpm_modules"] = {
        "inspection_exit": modules.returncode,
        "systemd-cryptsetup": "systemd-cryptsetup" in modules.stdout.split(),
        "tpm2-tss": "tpm2-tss" in modules.stdout.split(),
    }
    return facts


def save_report(report):
    root_directory(REPORT_DIR)
    report["timestamp_utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    report["script_sha256"] = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    history = REPORT_DIR / ("attempt-" + str(time.time_ns()) + ".json")
    with history.open("x") as stream:
        os.fchmod(stream.fileno(), 0o644)
        json.dump(report, stream, sort_keys=True, indent=2)
        stream.write("\n")
    if REPORT.exists() or REPORT.is_symlink():
        info = REPORT.lstat()
        require(
            stat.S_ISREG(info.st_mode) and info.st_uid == 0 and info.st_nlink == 1,
            "unsafe-report-file",
        )
    fd, name = tempfile.mkstemp(prefix=".report-", dir=REPORT_DIR)
    try:
        with os.fdopen(fd, "w") as stream:
            os.fchmod(stream.fileno(), 0o644)
            json.dump(report, stream, sort_keys=True, indent=2)
            stream.write("\n")
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(name, REPORT)
    finally:
        Path(name).unlink(missing_ok=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--apply", action="store_true")
    args = parser.parse_args()
    swaps = inspect_inputs()
    if not args.apply:
        print(
            json.dumps(
                {
                    "status": "PREVIEW",
                    "target_home_uuid": HOME_UUID,
                    "changes": [
                        "install missing Restic and required dependencies",
                        "disable default zram0 swap persistently",
                    ],
                    "active_swap_devices": len(swaps),
                    "root_report": str(REPORT),
                }
            )
        )
        return 0
    require(os.geteuid() == 0, "private-sudo-step-required")
    fd = os.open(
        "/run/hermes-production-prerequisites.lock",
        os.O_CREAT | os.O_RDWR | os.O_NOFOLLOW | os.O_CLOEXEC,
        0o600,
    )
    with os.fdopen(fd, "w") as lock:
        info = os.fstat(lock.fileno())
        require(
            stat.S_ISREG(info.st_mode)
            and info.st_uid == 0
            and info.st_nlink == 1
            and not info.st_mode & 0o077,
            "unsafe-operation-lock",
        )
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        swaps = inspect_inputs()
        if not REPORT_DIR.exists():
            REPORT_DIR.mkdir(mode=0o755)
            REPORT_DIR.chmod(0o755)
        root_directory(REPORT_DIR)
        report = {"status": "FAIL", "scope": "Restic/zram prerequisites only", "stage": "restic"}
        try:
            print("Installing Restic if missing; this can take several minutes.", flush=True)
            if run(["rpm", "-q", "restic"], check=False).returncode != 0:
                # Let DNF retain control of transaction cancellation and its RPM lock.
                run(
                    ["dnf", "-y", "--setopt=install_weak_deps=False", "install", "restic"],
                    timeout=None,
                )
            report["restic"] = run(["restic", "version"]).stdout.strip()
            report["stage"] = "zram"
            # Recheck memory pressure after package installation, before stopping swap.
            swaps = inspect_inputs()
            if not MASK.exists() and not MASK.is_symlink():
                os.symlink("/dev/null", MASK)
            if swaps:
                run(["systemctl", "stop", "dev-zram0.swap"])
            state = run(
                [
                    "systemctl",
                    "show",
                    "-p",
                    "ActiveState",
                    "--value",
                    "systemd-zram-setup@zram0.service",
                ]
            ).stdout.strip()
            if state == "active":
                run(["systemctl", "stop", "systemd-zram-setup@zram0.service"])
            else:
                require(state == "inactive", "zram-setup-state-needs-review")
            run(["systemctl", "daemon-reload"])
            require(
                not run(["swapon", "--show", "--noheadings"]).stdout.strip(), "swap-still-active"
            )
            require(
                not Path("/run/systemd/generator/dev-zram0.swap").exists(), "zram-unit-regenerated"
            )
            report.update(stage="inspection", active_swap=False, zram_persistent_override=True)
            report["root_inspection"] = root_facts()
            report.update(
                status="PASS",
                stage="complete",
                reboot="NOT_RUN",
                recovery_proofs="NOT_RUN",
                production_deployment="NOT_RUN",
            )
        except Exception as error:
            report["error"] = (
                str(error) if isinstance(error, RuntimeError) else type(error).__name__
            )
        save_report(report)
        print(json.dumps(report, sort_keys=True, indent=2))
        return int(report["status"] != "PASS")


if __name__ == "__main__":
    try:
        sys.exit(main())
    except Exception as error:
        print(
            json.dumps(
                {
                    "status": "REFUSED",
                    "reason": str(error)
                    if isinstance(error, RuntimeError)
                    else type(error).__name__,
                }
            ),
            file=sys.stderr,
        )
        sys.exit(1)
