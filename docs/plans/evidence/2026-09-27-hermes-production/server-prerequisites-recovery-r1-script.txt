"""Bounded prerequisites for aicowork@10.0.30.10; preview unless --apply."""

import argparse
import datetime
import fcntl
import hashlib
import json
import os
import stat
import subprocess
import sys
import tempfile
import time
from pathlib import Path

HOME_UUID = "eb129d35-1f52-4577-99e5-25daa322febc"
LUKS_UUID = "8f5a57cb-f1d7-4eb7-8d1f-266d65f8cff6"
PROC_SWAPS = Path("/proc/swaps")
ZRAM_SYSFS = Path("/sys/block/zram0")
ZRAM_DEVICE = Path("/dev/zram0")
MOUNTINFO = Path("/proc/self/mountinfo")
GENERATOR_DIRS = tuple(
    Path("/run/systemd") / name for name in ("generator", "generator.early", "generator.late")
)
SETUP_UNIT = "systemd-zram-setup@zram0.service"
SWAP_UNIT = "dev-zram0.swap"
MASK = Path("/etc/systemd/zram-generator.conf")
REPORT_DIR = Path("/var/lib/hermes-production-prerequisites")
REPORT = REPORT_DIR / "report.json"


def require(value, reason):
    if not value:
        raise RuntimeError(reason)


def run(args, *, check=True, timeout=60):
    result = subprocess.run(
        args,
        capture_output=True,
        text=True,
        timeout=timeout,
        env={"PATH": "/usr/sbin:/usr/bin:/sbin:/bin", "LANG": "C.UTF-8"},
        stdin=subprocess.DEVNULL,
    )
    if check:
        require(result.returncode == 0, "command-failed:" + Path(args[0]).name)
    return result


def root_directory(path):
    info = path.lstat()
    require(
        stat.S_ISDIR(info.st_mode) and info.st_uid == 0 and not info.st_mode & 0o022,
        "unsafe-root-directory",
    )


def swap_inventory(raw):
    rows = raw.splitlines()
    require(
        rows and rows[0].split() == ["Filename", "Type", "Size", "Used", "Priority"],
        "unexpected-swap-header",
    )
    swaps = []
    for row in rows[1:]:
        fields = row.split()
        require(
            len(fields) == 5 and fields[2].isdigit() and fields[3].isdigit(),
            "unexpected-swap-inventory",
        )
        swaps.append({"name": fields[0], "used": int(fields[3]) * 1024})
    return swaps


def inspect_inputs():
    require(
        run(["findmnt", "-n", "-o", "UUID", "--target", "/home/hermes"]).stdout.strip()
        == HOME_UUID,
        "wrong-target-filesystem",
    )
    for parent in (
        Path("/etc"),
        Path("/etc/systemd"),
        Path("/run"),
        Path("/var"),
        Path("/var/lib"),
    ):
        root_directory(parent)
    for item in Path("/proc/cmdline").read_text().split():
        require(
            item
            not in (
                "systemd.zram",
                "systemd.zram=1",
                "systemd.zram=yes",
                "systemd.zram=true",
                "systemd.zram=on",
            ),
            "kernel-forces-zram",
        )
    for prefix in ("/run", "/etc", "/usr/local/lib", "/usr/lib"):
        directory = Path(prefix) / "systemd/zram-generator.conf.d"
        require(not directory.is_symlink(), "zram-dropin-directory-symlink")
        require(
            not directory.exists() or not any(directory.glob("*.conf")),
            "existing-zram-dropins-require-review",
        )
    require(
        not Path("/run/systemd/zram-generator.conf").exists()
        and not Path("/run/systemd/zram-generator.conf").is_symlink(),
        "runtime-zram-configuration-conflict",
    )
    if MASK.is_symlink():
        require(
            os.readlink(MASK) == "/dev/null" and MASK.lstat().st_uid == 0,
            "existing-zram-mask-conflict",
        )
    elif MASK.exists():
        info = MASK.lstat()
        require(
            stat.S_ISREG(info.st_mode)
            and info.st_uid == 0
            and info.st_nlink == 1
            and not info.st_mode & 0o022
            and MASK.read_bytes() == b"",
            "existing-zram-configuration-conflict",
        )
    swaps = swap_inventory(PROC_SWAPS.read_text())
    require(all(row["name"] == "/dev/zram0" for row in swaps), "unrelated-swap-requires-review")
    available = next(
        int(line.split()[1]) * 1024
        for line in Path("/proc/meminfo").read_text().splitlines()
        if line.startswith("MemAvailable:")
    )
    require(
        available >= sum(int(row["used"]) for row in swaps) + 2 * 1024**3,
        "insufficient-memory-for-swapoff",
    )
    return swaps


def unit_state(unit):
    result = run(["systemctl", "show", "-p", "LoadState", "-p", "ActiveState", unit])
    values = dict(line.split("=", 1) for line in result.stdout.splitlines() if "=" in line)
    require(values.get("LoadState") in ("loaded", "not-found"), "unexpected-zram-unit-load-state")
    require(bool(values.get("ActiveState")), "missing-zram-unit-state")
    return values


def device_state():
    if not ZRAM_SYSFS.exists():
        require(not ZRAM_DEVICE.exists(), "zram-device-without-sysfs")
        return {"present": False, "initialized": 0, "size_bytes": 0}
    info = ZRAM_DEVICE.lstat()
    require(stat.S_ISBLK(info.st_mode), "unexpected-zram-device-type")
    identity = f"{os.major(info.st_rdev)}:{os.minor(info.st_rdev)}"
    require((ZRAM_SYSFS / "dev").read_text().strip() == identity, "zram-device-identity-mismatch")
    require(not any((ZRAM_SYSFS / "holders").iterdir()), "zram-device-has-holders")
    require(not any(ZRAM_SYSFS.glob("zram0p*")), "zram-device-has-partitions")
    require((ZRAM_SYSFS / "backing_dev").read_text().strip() == "none", "zram-has-backing-device")
    for line in MOUNTINFO.read_text().splitlines():
        fields = line.split()
        require(len(fields) >= 6, "unexpected-mount-inventory")
        require(fields[2] != identity, "zram-device-mounted")
    initialized = (ZRAM_SYSFS / "initstate").read_text().strip()
    size = (ZRAM_SYSFS / "disksize").read_text().strip()
    require(initialized in ("0", "1") and size.isdigit(), "unexpected-zram-device-state")
    return {"present": True, "initialized": int(initialized), "size_bytes": int(size)}


def disable_zram():
    # Recheck target, configuration and memory after the package transaction.
    swaps = inspect_inputs()
    before = device_state()
    initial_units = {unit: unit_state(unit) for unit in (SWAP_UNIT, SETUP_UNIT)}
    if swaps:
        # Finish kernel swapoff before a setup-service stop can attempt reset.
        run(["swapoff", str(ZRAM_DEVICE)])
    require(not swap_inventory(PROC_SWAPS.read_text()), "swap-still-active")
    loaded = [unit for unit, state in initial_units.items() if state["LoadState"] == "loaded"]
    if loaded:
        run(["systemctl", "stop", *loaded])
    stopped_units = {unit: unit_state(unit) for unit in (SWAP_UNIT, SETUP_UNIT)}
    require(
        all(state["ActiveState"] in ("inactive", "failed") for state in stopped_units.values()),
        "zram-units-not-stopped",
    )
    require(not swap_inventory(PROC_SWAPS.read_text()), "swap-still-active")
    current = device_state()
    reset_needed = bool(current["initialized"] or current["size_bytes"])
    if reset_needed:
        # The kernel refuses reset if an opener or another namespace is still using it.
        run(["zramctl", "--reset", str(ZRAM_DEVICE)])
    after = device_state()
    require(not after["initialized"] and after["size_bytes"] == 0, "zram-device-not-reset")
    # A failed marker is cleared only after successful, independently verified teardown.
    failed = [unit for unit, state in stopped_units.items() if state["ActiveState"] == "failed"]
    if failed:
        run(["systemctl", "reset-failed", *failed])
    if not MASK.exists() and not MASK.is_symlink():
        os.symlink("/dev/null", MASK)
    run(["systemctl", "daemon-reload"])
    require(not inspect_inputs(), "swap-still-active")
    for directory in GENERATOR_DIRS:
        require(not (directory / SWAP_UNIT).exists(), "zram-unit-regenerated")
        require(not (directory / (SETUP_UNIT + ".d")).exists(), "zram-bindings-regenerated")
    final_units = {unit: unit_state(unit) for unit in (SWAP_UNIT, SETUP_UNIT)}
    require(
        all(state["ActiveState"] == "inactive" for state in final_units.values()),
        "zram-units-not-inactive",
    )
    after = device_state()
    require(not after["initialized"] and after["size_bytes"] == 0, "zram-device-not-reset")
    return {
        "before": before,
        "after": after,
        "initial_units": initial_units,
        "final_units": final_units,
        "explicit_device_reset": reset_needed,
        "cleared_failed_units": failed,
    }


def root_facts():
    facts = {}
    for name, args in {
        "logical_volumes": [
            "lvs",
            "hermes_system",
            "--reportformat",
            "json",
            "--units",
            "b",
            "--nosuffix",
            "-o",
            "vg_name,lv_name,segtype,lv_size",
        ],
        "volume_group": [
            "vgs",
            "hermes_system",
            "--reportformat",
            "json",
            "--units",
            "b",
            "--nosuffix",
            "-o",
            "vg_name,vg_size,vg_free",
        ],
    }.items():
        value = run(args, check=False)
        facts[name] = (
            json.loads(value.stdout) if value.returncode == 0 else {"status": "UNVERIFIED"}
        )
    metadata = run(
        ["cryptsetup", "luksDump", "--dump-json-metadata", "/dev/disk/by-uuid/" + LUKS_UUID],
        check=False,
    )
    if metadata.returncode == 0:
        value = json.loads(metadata.stdout)
        facts["luks2"] = {
            "keyslot_count": len(value.get("keyslots", {})),
            "tpm_tokens": [
                {
                    "pcrs": token.get("tpm2-pcrs"),
                    "bank": token.get("tpm2-pcr-bank"),
                    "pin": token.get("tpm2-pin", False),
                }
                for token in value.get("tokens", {}).values()
                if token.get("type") == "systemd-tpm2"
            ],
        }
    else:
        facts["luks2"] = {"status": "UNVERIFIED"}
    crypttab = Path("/etc/crypttab")
    facts["crypttab"] = []
    if crypttab.exists():
        for line in crypttab.read_text().splitlines():
            fields = line.split()
            if len(fields) >= 2 and fields[1] == "UUID=" + LUKS_UUID:
                facts["crypttab"].append(
                    {
                        "keyfile_bypass": len(fields) > 2 and fields[2] not in ("none", "-"),
                        "tpm2_auto": len(fields) > 3 and "tpm2-device=auto" in fields[3].split(","),
                    }
                )
    modules = run(["lsinitrd", "-m", "/boot/initramfs-" + os.uname().release + ".img"], check=False)
    facts["initramfs_tpm_modules"] = {
        "inspection_exit": modules.returncode,
        "systemd-cryptsetup": "systemd-cryptsetup" in modules.stdout.split(),
        "tpm2-tss": "tpm2-tss" in modules.stdout.split(),
    }
    return facts


def save_report(report):
    root_directory(REPORT_DIR)
    report["timestamp_utc"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    report["script_sha256"] = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    history = REPORT_DIR / ("attempt-" + str(time.time_ns()) + ".json")
    with history.open("x") as stream:
        os.fchmod(stream.fileno(), 0o644)
        json.dump(report, stream, sort_keys=True, indent=2)
        stream.write("\n")
    if REPORT.exists() or REPORT.is_symlink():
        info = REPORT.lstat()
        require(
            stat.S_ISREG(info.st_mode) and info.st_uid == 0 and info.st_nlink == 1,
            "unsafe-report-file",
        )
    fd, name = tempfile.mkstemp(prefix=".report-", dir=REPORT_DIR)
    try:
        with os.fdopen(fd, "w") as stream:
            os.fchmod(stream.fileno(), 0o644)
            json.dump(report, stream, sort_keys=True, indent=2)
            stream.write("\n")
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(name, REPORT)
    finally:
        Path(name).unlink(missing_ok=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--apply", action="store_true")
    args = parser.parse_args()
    swaps = inspect_inputs()
    if not args.apply:
        print(
            json.dumps(
                {
                    "status": "PREVIEW",
                    "target_home_uuid": HOME_UUID,
                    "changes": [
                        "install missing Restic and required dependencies",
                        "disable default zram0 swap persistently",
                    ],
                    "active_swap_devices": len(swaps),
                    "zram_device": device_state(),
                    "zram_units": {unit: unit_state(unit) for unit in (SWAP_UNIT, SETUP_UNIT)},
                    "zramctl": run(["zramctl", "--version"]).stdout.strip(),
                    "root_report": str(REPORT),
                }
            )
        )
        return 0
    require(os.geteuid() == 0, "private-sudo-step-required")
    fd = os.open(
        "/run/hermes-production-prerequisites.lock",
        os.O_CREAT | os.O_RDWR | os.O_NOFOLLOW | os.O_CLOEXEC,
        0o600,
    )
    with os.fdopen(fd, "w") as lock:
        info = os.fstat(lock.fileno())
        require(
            stat.S_ISREG(info.st_mode)
            and info.st_uid == 0
            and info.st_nlink == 1
            and not info.st_mode & 0o077,
            "unsafe-operation-lock",
        )
        fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        swaps = inspect_inputs()
        if not REPORT_DIR.exists():
            REPORT_DIR.mkdir(mode=0o755)
            REPORT_DIR.chmod(0o755)
        root_directory(REPORT_DIR)
        report = {"status": "FAIL", "scope": "Restic/zram prerequisites only", "stage": "restic"}
        try:
            print("Installing Restic if missing; this can take several minutes.", flush=True)
            if run(["rpm", "-q", "restic"], check=False).returncode != 0:
                # Let DNF retain control of transaction cancellation and its RPM lock.
                run(
                    ["dnf", "-y", "--setopt=install_weak_deps=False", "install", "restic"],
                    timeout=None,
                )
            report["restic"] = run(["restic", "version"]).stdout.strip()
            report["stage"] = "zram"
            report["zram_recovery"] = disable_zram()
            report.update(stage="inspection", active_swap=False, zram_persistent_override=True)
            report["root_inspection"] = root_facts()
            report.update(
                status="PASS",
                stage="complete",
                reboot="NOT_RUN",
                recovery_proofs="NOT_RUN",
                production_deployment="NOT_RUN",
            )
        except Exception as error:
            report["error"] = (
                str(error) if isinstance(error, RuntimeError) else type(error).__name__
            )
        save_report(report)
        print(json.dumps(report, sort_keys=True, indent=2))
        return int(report["status"] != "PASS")


if __name__ == "__main__":
    try:
        sys.exit(main())
    except Exception as error:
        print(
            json.dumps(
                {
                    "status": "REFUSED",
                    "reason": str(error)
                    if isinstance(error, RuntimeError)
                    else type(error).__name__,
                }
            ),
            file=sys.stderr,
        )
        sys.exit(1)
