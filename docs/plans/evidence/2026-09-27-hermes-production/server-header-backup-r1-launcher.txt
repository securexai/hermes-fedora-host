#!/usr/bin/env bash
set -Eeuo pipefail

if [[ ! -t 0 || ! -t 1 ]]; then
  printf 'Run this command in your interactive terminal for private sudo entry.\n' >&2
  exit 1
fi

recovery_dir="$(cd -- "$(dirname -- "${BASH_SOURCE[0]}")" && pwd)"
access_dir="$(cd -- "$recovery_dir/../hermes-production-access" && pwd)"
ssh -t -F /dev/null \
  -i "$access_dir/aicowork-10.0.30.10" \
  -o "UserKnownHostsFile=$access_dir/known_hosts" \
  -o GlobalKnownHostsFile=/dev/null \
  -o StrictHostKeyChecking=yes \
  -o UpdateHostKeys=no \
  -o BatchMode=yes \
  -o IdentitiesOnly=yes \
  -o IdentityAgent=none \
  -o ForwardAgent=no \
  -o ClearAllForwardings=yes \
  -o ConnectTimeout=8 \
  aicowork@10.0.30.10 \
  'sudo python3 -I /home/aicowork/hermes-header-backup-db2ceeae744d230d.py --apply'
python3 -I "$recovery_dir/fetch-and-verify.py"
